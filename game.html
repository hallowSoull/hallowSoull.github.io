The Consumerist Fly - A Minigame!

Use your cursor to move the fly around and sit on the food on the table to level up

We often criticize the fly for seeking to do nothing but consume, but at the end of the day, don't we all do the same?

Here's the code! 

var ballx = 300;
var bally = 300;
var ballSize = 200;
var score =0;
var gameState= "L1";
var img1, img2, img3, img4, img5, img6, img7;

function preload() {

  img1 = loadImage('https://hallowsoull.github.io/fly_minigame/table.png');

function setup() {
  createCanvas(600, 600);
  textAlign(CENTER);
  textSize(20);
} // end setup


function draw() {
   // background(220);
   background(img1);
  if (gameState=="L1"){
  levelOne();
  } 
  if (gameState=="L2"){
   levelTwo(); 
  }
  if (gameState=="L3"){
   levelThree(); 
  }
  if (gameState=="L4"){
  levelFour();
  } 
  if (gameState=="L5"){
   levelFive(); 
  }
  
  
  rect (322, 38, 19, 21);
  text(("Score: " + score), width/2, 55);
  text("Become a fly and sit on as much food as you can!", width/2, 30);
  

} // end draw
